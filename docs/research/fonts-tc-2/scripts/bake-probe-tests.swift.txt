import CoreText
import SwiftUI
import UIKit
import XCTest
@testable import DaysCam

/// TASK-142 142.4 probe (#494): for the heaviest face of each of the 30
/// families, does the whole overlay line draw in that font, in the viewfinder
/// and in the baked photo alike (method of `docs/research/fonts-tc.md` §C)?
///
/// Not part of the regular suite: skips unless `DAYSCAM_142_PROBE=1`. Needs
/// the font files (`DAYSCAM_142_FONTS`, one folder holding the files of
/// `dayscam-fonts@v4`) and an output folder (`DAYSCAM_142_OUT`). Pass the
/// variables with the `TEST_RUNNER_` prefix. Results go to
/// `<out>/probe.tsv` and PNG sheets, because xcodebuild hides `print`.
@MainActor
final class Bake142ProbeTests: XCTestCase {
    static let text = "2026.10.11 我的紀錄 第 128 天"

    private struct AllAvailable: FontAvailabilityProviding {
        func state(for fontID: OverlayFontID) -> FontState { .available }
        func state(for fontID: OverlayFontID, weightID: String?) -> FontState { .available }
    }

    private var environment: [String: String] { ProcessInfo.processInfo.environment }

    func test_probe_heaviestFacesDrawTheWholeLineInTheirOwnFont() throws {
        try XCTSkipUnless(environment["DAYSCAM_142_PROBE"] == "1", "set DAYSCAM_142_PROBE=1")
        let fontsDir = URL(fileURLWithPath: try XCTUnwrap(environment["DAYSCAM_142_FONTS"]))
        let outDir = URL(fileURLWithPath: try XCTUnwrap(environment["DAYSCAM_142_OUT"]))
        try FileManager.default.createDirectory(at: outDir, withIntermediateDirectories: true)

        let fontSize = OverlayMetrics.baseFontSize(forShortEdge: 1200)
        let pingFangImage = baked(font: .pingFang, size: fontSize)
        var rows: [(label: String, baked: UIImage, preview: UIImage)] = []
        var report = ["fontID\tweight\tpostScriptName\tresolved\trunFonts\tmissing\tbakedInk\tpreviewInk\tpingFangInk"]
        let pingFangInk = inkWidth(of: pingFangImage)

        for family in FontCatalog.downloadFamilies {
            let face = family.faces[family.faces.count - 1]
            let url = fontsDir.appendingPathComponent(face.asset.fileName)
            var error: Unmanaged<CFError>?
            CTFontManagerRegisterFontsForURL(url as CFURL, .process, &error)
            error?.release()

            let resolved = OverlayMetrics.resolvedFont(for: family.fontID, weightID: face.id,
                                                       availability: AllAvailable())
            let font = resolved.font(ofSize: fontSize)
            let runs = usedFonts(for: Self.text, font: font)
            let runFonts = Array(NSOrderedSet(array: runs.map(\.font))) as? [String] ?? []
            let missing = runs.filter { $0.font != face.postScriptName }.map(\.text).joined()
            let bakedImage = baked(font: resolved, size: fontSize)
            let previewImage = preview(font: font)
            let bakedInk = inkWidth(of: bakedImage)
            let previewInk = inkWidth(of: previewImage)
            report.append([family.fontID.rawValue, face.id, face.postScriptName, resolved.postScriptName,
                           runFonts.joined(separator: ","), missing.isEmpty ? "-" : missing,
                           "\(bakedInk)", "\(previewInk)", "\(pingFangInk)"].joined(separator: "\t"))
            rows.append(("\(family.displayName)  \(face.postScriptName)", bakedImage, previewImage))
            XCTAssertEqual(resolved.postScriptName, face.postScriptName, "\(family.fontID) must not fall back")
            if missing.isEmpty {
                XCTAssertEqual(runFonts, [face.postScriptName], "\(family.fontID) draws the whole line itself")
            }
            XCTAssertNotEqual(bakedInk, pingFangInk, "\(family.fontID): same ink width as PingFang?")
            XCTAssertEqual(bakedInk, previewInk, accuracy: 2, "\(family.fontID): viewfinder and bake differ")
        }
        try report.joined(separator: "\n").write(to: outDir.appendingPathComponent("probe.tsv"),
                                                 atomically: true, encoding: .utf8)
        // Sheets of ten rows: PingFang reference first, then baked | viewfinder.
        for (index, chunk) in stride(from: 0, to: rows.count, by: 10).map({ Array(rows[$0..<min($0 + 10, rows.count)]) }).enumerated() {
            let sheet = sheetImage(reference: pingFangImage, rows: chunk)
            try sheet.pngData()?.write(to: outDir.appendingPathComponent("bake-sheet-\(index + 1).png"))
        }
    }

    /// 未來熒黑 lacks 值告填巔慎教清真鎮顛: record which font draws them.
    func test_probe_glowSansMissingCharacters() throws {
        try XCTSkipUnless(environment["DAYSCAM_142_PROBE"] == "1", "set DAYSCAM_142_PROBE=1")
        let fontsDir = URL(fileURLWithPath: try XCTUnwrap(environment["DAYSCAM_142_FONTS"]))
        let outDir = URL(fileURLWithPath: try XCTUnwrap(environment["DAYSCAM_142_OUT"]))
        var error: Unmanaged<CFError>?
        CTFontManagerRegisterFontsForURL(fontsDir.appendingPathComponent("GlowSansTC-Normal-Heavy.otf") as CFURL, .process, &error)
        let resolved = OverlayMetrics.resolvedFont(for: .glowSans, weightID: "black", availability: AllAvailable())
        let font = resolved.font(ofSize: 40)
        let runs = usedFonts(for: "值告填巔慎教清真鎮顛 第 128 天", font: font)
        let line = runs.map { "\($0.text)=\($0.font)" }.joined(separator: "|")
        try line.write(to: outDir.appendingPathComponent("glow-missing.txt"), atomically: true, encoding: .utf8)
        XCTAssertEqual(resolved.postScriptName, "GlowSansTC-Normal-Heavy")
    }

    // MARK: Helpers

    /// A dark 1200 x 1600 photo with the line baked by the real renderer,
    /// cropped to the band the line sits in.
    private func baked(font: ResolvedOverlayFont, size: CGFloat) -> UIImage {
        let format = UIGraphicsImageRendererFormat()
        format.scale = 1
        let photo = UIGraphicsImageRenderer(size: CGSize(width: 1200, height: 1600), format: format).image { context in
            UIColor(white: 0.12, alpha: 1).setFill()
            context.fill(CGRect(x: 0, y: 0, width: 1200, height: 1600))
        }
        let style = OverlayStyle(fontID: .pingFang, textColor: .white, outline: .none, shadow: .none)
        let result = OverlayRenderer().bake(image: photo, text: Self.text, style: style,
                                            placement: .anchor(.bottomLeading), font: font) ?? photo
        guard let cg = result.cgImage?.cropping(to: CGRect(x: 0, y: 1400, width: 1200, height: 200)) else { return result }
        return UIImage(cgImage: cg)
    }

    @MainActor
    private func preview(font: UIFont) -> UIImage {
        let content = Text(Self.text)
            .font(Font(font))
            .foregroundColor(.white)
            .fixedSize()
            .padding(.horizontal, 20)
            .frame(width: 1200, height: 200, alignment: .leading)
            .background(Color(white: 0.12))
        let renderer = ImageRenderer(content: content)
        renderer.scale = 1
        return renderer.uiImage ?? UIImage()
    }

    private func usedFonts(for text: String, font: UIFont) -> [(text: String, font: String)] {
        let line = CTLineCreateWithAttributedString(NSAttributedString(string: text, attributes: [.font: font]))
        let runs = CTLineGetGlyphRuns(line) as? [CTRun] ?? []
        let string = text as NSString
        return runs.map { run in
            let range = CTRunGetStringRange(run)
            let attributes = CTRunGetAttributes(run) as NSDictionary
            let name = (attributes[kCTFontAttributeName] as? UIFont)?.fontName ?? "?"
            return (string.substring(with: NSRange(location: range.location, length: range.length)), name)
        }
    }

    /// Width of the box around all bright pixels.
    private func inkWidth(of image: UIImage) -> Int {
        guard let cg = image.cgImage else { return -1 }
        let width = cg.width, height = cg.height
        var pixels = [UInt8](repeating: 0, count: width * height * 4)
        guard let context = CGContext(data: &pixels, width: width, height: height, bitsPerComponent: 8,
                                      bytesPerRow: width * 4, space: CGColorSpaceCreateDeviceRGB(),
                                      bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return -1 }
        context.draw(cg, in: CGRect(x: 0, y: 0, width: width, height: height))
        var minX = width, maxX = -1
        for y in 0..<height {
            for x in 0..<width where pixels[(y * width + x) * 4] > 160 {
                minX = min(minX, x)
                maxX = max(maxX, x)
            }
        }
        return maxX >= minX ? maxX - minX + 1 : 0
    }

    private func sheetImage(reference: UIImage, rows: [(label: String, baked: UIImage, preview: UIImage)]) -> UIImage {
        let rowHeight: CGFloat = 200, labelHeight: CGFloat = 44, width: CGFloat = 1200
        let all = [("蘋方 PingFangTC-Semibold（對照）", reference, reference)] + rows
        let height = CGFloat(all.count) * (rowHeight * 2 + labelHeight)
        let format = UIGraphicsImageRendererFormat()
        format.scale = 1
        return UIGraphicsImageRenderer(size: CGSize(width: width, height: height), format: format).image { context in
            UIColor(white: 0.12, alpha: 1).setFill()
            context.fill(CGRect(x: 0, y: 0, width: width, height: height))
            var y: CGFloat = 0
            for row in all {
                (row.0 + "　上：烘焙　下：觀景窗" as NSString).draw(
                    at: CGPoint(x: 12, y: y + 8),
                    withAttributes: [.font: UIFont.systemFont(ofSize: 26, weight: .medium),
                                     .foregroundColor: UIColor.systemYellow])
                y += labelHeight
                row.1.draw(at: CGPoint(x: 0, y: y))
                y += rowHeight
                row.2.draw(at: CGPoint(x: 0, y: y))
                y += rowHeight
            }
        }
    }
}
