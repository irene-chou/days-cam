import XCTest

/// TASK-142 B′ shots (#494): the font list with the 30 new families (not yet
/// downloaded, so their preview images show) and the license page. Writes PNGs
/// to `DAYSCAM_UI_SHOT_DIR`. Throwaway; not part of the suite.
final class Shots142UITests: XCTestCase {
    private var app: XCUIApplication!

    override func setUp() {
        super.setUp()
        continueAfterFailure = false
        app = XCUIApplication()
        app.launchArguments += ["-UITestSkipOnboarding", "-UITestFreshProfiles", UUID().uuidString]
        app.launch()
        XCTAssertTrue(app.buttons["疊字設定"].waitForExistence(timeout: 10))
        if !app.buttons["樣式"].exists { app.buttons["疊字設定"].tap() }
        XCTAssertTrue(app.buttons["樣式"].waitForExistence(timeout: 5))
        app.buttons["樣式"].tap()
        app.buttons["字型"].tap()
    }

    private func shot(_ name: String) {
        let screenshot = app.screenshot()
        let attachment = XCTAttachment(screenshot: screenshot)
        attachment.name = name
        attachment.lifetime = .keepAlways
        add(attachment)
        if let path = ProcessInfo.processInfo.environment["DAYSCAM_UI_SHOT_DIR"], !path.isEmpty {
            let directory = URL(fileURLWithPath: path, isDirectory: true)
            try? FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
            try? screenshot.pngRepresentation.write(to: directory.appendingPathComponent("\(name).png"))
        }
    }

    func testFontListAndLicensePage() {
        let pingFang = app.buttons["蘋方-繁"]
        XCTAssertTrue(pingFang.waitForExistence(timeout: 5))
        let listTop = pingFang.frame.minY
        let height = app.windows.firstMatch.frame.height
        let listBottom = height - (height > 800 ? 50 : 20)
        let origin = app.coordinate(withNormalizedOffset: .zero)
        func drag(_ times: Int, _ velocity: XCUIGestureVelocity) {
            for _ in 0..<times {
                origin.withOffset(CGVector(dx: 60, dy: listBottom))
                    .press(forDuration: 0.1, thenDragTo: origin.withOffset(CGVector(dx: 60, dy: listTop + 6)),
                           withVelocity: velocity, thenHoldForDuration: 0.2)
            }
        }
        shot("142_list_01_top")
        for index in 0..<4 {
            drag(4, .slow)
            shot("142_list_0\(index + 2)")
        }
        drag(40, .fast)
        let license = app.buttons["字型授權"]
        XCTAssertTrue(license.waitForExistence(timeout: 5))
        shot("142_list_08_bottom")
        license.tap()
        XCTAssertTrue(app.staticTexts["字型授權"].waitForExistence(timeout: 5))
        shot("142_license_01_top")
        for index in 0..<4 {
            for _ in 0..<14 {
                origin.withOffset(CGVector(dx: 60, dy: height - 120))
                    .press(forDuration: 0.1, thenDragTo: origin.withOffset(CGVector(dx: 60, dy: 140)),
                           withVelocity: .fast, thenHoldForDuration: 0.1)
            }
            shot("142_license_0\(index + 2)")
        }
    }
}
